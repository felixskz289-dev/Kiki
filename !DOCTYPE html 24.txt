function startBuild(){ bIdx=0; built=[]; renderBuild(); }

  html+='</div><div id="bresult" style="margin-top:12px;font-size:14px;min-height:20px"></div>';

    html+='<button class="btn" style="width:100%;margin-top:8px" onclick="bcheck()">Проверить</button>';
  }
  html+='<div class="nav"><button class="btn ghost" onclick="breset()">Сброс</button>'+(bIdx<BUILDERS.length-1?'<button class="btn" onclick="bnext()">Следующее →</button>':'<button class="btn" onclick="startBuild()">Сначала ↺</button>')+'</div></div>';
  document.getElementById('build-body').innerHTML=html;
}
function badd(t,i){ built.push(t); const el=document.getElementById('bp'+i); if(el) el.style.display='none'; renderBuild(); }
function btokenRemove(i){ built.splice(i,1); renderBuild(); }
function breset(){ built=[]; renderBuild(); }
function bcheck(){
  const b=BUILDERS[bIdx]; const joined=built.join('');

    : '<span style="color:#E0442E;font-weight:700">✗ Попробуй ещё</span>';
}
function bnext(){ bIdx++; built=[]; renderBuild(); }
